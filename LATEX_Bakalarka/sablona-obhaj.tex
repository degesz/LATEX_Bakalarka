\begin{frame} 
	% nadpis snímku
	\frametitle{Zadání práce}
	
	\begin{enumerate}
		\item Nastudujte principy měření impedance elektrických součástek a dostupné metody realizace měřicích přístrojů v~oblasti nízkých a~středních frekvencí. Zaměřte se na způsoby buzení měřeného prvku, zpracování signálu a~vyhodnocení výsledků.
		
		\item Na základě rešerše navrhněte schéma přístroje pro měření impedance v~pásmu nastavitelných frekvencí do 100~kHz. Přístroj by měl umožňovat charakterizaci pasivních součástek, zejména cívek a~kondenzátorů.
		
		\item Výsledky zpracujte v~semestrálním projektu.
	\end{enumerate}
	
	\vspace{2ex}
	\begin{block}{Specifikace navrhovaného zařízení}
		\begin{itemize}
			\item Frekvenční rozsah: $100~\mathrm{Hz}$ -- $100~\mathrm{kHz}$
			\item Cílová přesnost: $1\,\%$
			\item Nastavitelná amplituda: $0{,}1~\mathrm{V}$ -- $2~\mathrm{V}$
			\item Napájení a komunikace přes USB
		\end{itemize}
	\end{block}
\end{frame}

\begin{frame} 
	\frametitle{Digitalizace signálů -- Přímé vzorkování}
	
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\textbf{Princip metody:}
			\begin{itemize}
				\item Synchronní vzorkování ADC
				\item Diskrétní Fourierova transformace
				\item Výpočet amplitudy a fáze v MCU
			\end{itemize}
			
			\vspace{1em}
			\begin{block}{Výhoda}
				Přesnější měření
			\end{block}
			
			\vspace{0.5em}
			\begin{alertblock}{Nevýhody}
				\begin{itemize}
					\item Nutnost rychlého vzorkování
					\item Složitější výpočet
				\end{itemize}
			\end{alertblock}
		\end{column}
		%
		\begin{column}{0.5\textwidth}
			\textbf{Výpočet parametrů:}
			$$A = \sqrt{U_c^2 + U_s^2}$$
			$$\varphi = \arctan\left(\frac{-U_s}{U_c}\right)$$
			
			\vspace{1em}
			kde $U_c$ a $U_s$ jsou kosinová a~sinová složka signálu z~DFT.
		\end{column}
	\end{columns}
\end{frame}

\begin{frame} 
	\frametitle{Zjištěné nedostatky prototypu}
	
	\begin{alertblock}{Identifikované problémy}
		\begin{enumerate}
			\item \textbf{Zkreslující OZ (TLV9154)} \\
			Při vyšších frekvencích zkresluje tvar sinusového signálu $\rightarrow$ chyba měření
			
			\vspace{0.5em}
			\item \textbf{Nadbytečné zesilovače} \\
			Diferenční zesilovače INA823 s přepínáním zesílení -- zbytečná komplikace
			
			\vspace{0.5em}
			\item \textbf{Nepřesné měření} \\
			Amplituda ze šp. detektoru a měření průchodu nulou neposkytovaly přesné výsledky
		\end{enumerate}
	\end{alertblock}
	
	\vspace{1em}
	\begin{block}{Závěr z testování}
		$\Rightarrow$ Přechod na metodu přímého vzorkování harmonických signálů
	\end{block}
